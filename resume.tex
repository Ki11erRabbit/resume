% ── Key-presence check ────────────────────────────────────────────────────────
%  \CheckKeys{include-list}{key,key,...}
%  Sets \ifsectionhascontent true if include-list is "all" OR any key is found.
%  Used to decide whether to emit a \section heading before its entries.
%
%  Implementation: walks the key list recursively using a sentinel token.
%  Each key is tested with \IfSubStr against the padded include-list.
%  The recursion stops when it reaches the sentinel or finds a match.

\newif\ifsectionhascontent

\long\def\checknextkey#1,#2\stopcheck#3{%
  \ifsectionhascontent\else
    \IfSubStr{,#3,}{,#1,}{\sectionhascontenttrue}{%
      \IfStrEq{#2}{}{}{\checknextkey#2\stopcheck{#3}}%
    }%
  \fi
}

\newcommand{\CheckKeys}[2]{%
  \sectionhascontentfalse%
  \IfStrEq{#1}{all}{\sectionhascontenttrue}{%
    \checknextkey#2,\stopcheck{#1}%
  }%
}

% ── Entry inclusion helper ────────────────────────────────────────────────────
%  \IncludeIfListed{key}{include-list}{file}
%  Inputs file if key is in include-list or include-list == "all".
\newcommand{\IncludeIfListed}[3]{%
  \IfStrEq{#2}{all}{%
    \input{#3}%
  }{%
    \IfSubStr{,#2,}{,#1,}{\input{#3}}{}%
  }%
}

% ── Reusable entry macros ─────────────────────────────────────────────────────
% \JobEntry{Title}{Org}{Date}{Location}
\newcommand{\JobEntry}[4]{%

}
